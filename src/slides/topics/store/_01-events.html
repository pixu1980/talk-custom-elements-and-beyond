emitChange(): the store speaks with CustomEvent

    
#emitChange(path, oldValue, newValue) {
  // A single, explicit event for every successful write.
  const detail = {
    path: pathToString(path),
    oldValue: deepClone(oldValue),
    newValue: deepClone(newValue),
  };

  // EventTarget keeps persistence, logging and the
  // custom elements decoupled from the store itself.
  this.events.dispatchEvent(
    new CustomEvent("store:change", { detail }),
  );
}
