Declarative form bindings

The store is the single source of truth: value / checked read it, @input / @change write it. No model directive, no hidden two-way magic. 
    
      <!-- input: value reads, @input writes -->
      <input value="${todo.title}"
        @input=${(e) => updateTodo(id, { 
          title: e.currentTarget.value 
        })} />

      <!-- checkbox: checked + @change -->
      <input type="checkbox" checked=${todo.completed}
        @change=${(e) => updateTodo(id, { 
          completed: e.currentTarget.checked 
        })} />

      <!-- select: data-value applied after the options render -->
      <select data-value="${todo.priority}"
      @change=${(e) => updateTodo(id, { 
          priority: e.currentTarget.value 
        })}>
        
      </select>
